============================================================
  VERIFICATION COMMANDS
============================================================

# Check all pods are running:
kubectl get pods -l app.kubernetes.io/instance={{ .Release.Name }}

# Check Dapr sidecars (if enabled):
{{- if .Values.dapr.enabled }}
kubectl get pods -l dapr.io/enabled=true
{{- end }}

# View logs:
kubectl logs -l app=backend --follow
kubectl logs -l app=frontend --follow
{{- if .Values.notificationService.enabled }}
kubectl logs -l app=notification --follow
{{- end }}
{{- if .Values.recurringService.enabled }}
kubectl logs -l app=recurring --follow
{{- end }}

# Health checks:
kubectl exec -it deploy/{{ include "todo-chatbot.fullname" . }}-backend -- curl -s localhost:8000/health
{{- if .Values.notificationService.enabled }}
kubectl exec -it deploy/{{ include "todo-chatbot.fullname" . }}-notification -- curl -s localhost:8001/health
{{- end }}
{{- if .Values.recurringService.enabled }}
kubectl exec -it deploy/{{ include "todo-chatbot.fullname" . }}-recurring -- curl -s localhost:8002/health
{{- end }}
